PRAGMA foreign_keys=OFF;
BEGIN TRANSACTION;

CREATE TABLE konsumen (
    id INTEGER PRIMARY KEY, 
    nama_lengkap TEXT NOT NULL, 
    email TEXT UNIQUE, 
    nomor_telp TEXT, 
    alamat TEXT, 
    last_seen TEXT, 
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE products (
    id INTEGER PRIMARY KEY,
    sku TEXT UNIQUE NOT NULL,
    name TEXT NOT NULL,
    stock INTEGER DEFAULT 0,
    is_active INTEGER DEFAULT 1 -- Boolean: 1 aktif, 0 non-aktif
);

CREATE TABLE price_levels (
    id INTEGER PRIMARY KEY ,
    level_name TEXT NOT NULL UNIQUE
);

INSERT INTO price_levels VALUES(1,'order');
INSERT INTO price_levels VALUES(2,'reseler');

CREATE TABLE product_prices (
    product_id INTEGER, 
    price_level_id INTEGER, 
    price INTEGER NOT NULL, 
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP, 
    PRIMARY KEY (product_id, price_level_id), 
    FOREIGN KEY (product_id) REFERENCES products (id) 
                        ON DELETE CASCADE, 
    FOREIGN KEY (price_level_id) REFERENCES price_levels (id) 
                        ON DELETE CASCADE
);

CREATE TABLE order_items (
    id INTEGER PRIMARY KEY,
    order_id INTEGER,
    product_id INTEGER,
    quantity INTEGER NOT NULL,
    base_price REAL NOT NULL,
    subtotal REAL NOT NULL,
    FOREIGN KEY (order_id) REFERENCES orders(id) ON DELETE CASCADE,
    FOREIGN KEY (product_id) REFERENCES products(id)
);

CREATE TABLE order_item_finishings (
    id INTEGER PRIMARY KEY,
    order_item_id INTEGER,
    finishing_id INTEGER,
    finishing_price REAL NOT NULL,
    FOREIGN KEY (order_item_id) REFERENCES order_items(id) ON DELETE CASCADE,
    FOREIGN KEY (finishing_id) REFERENCES finishing_services(id)
);

CREATE TABLE roles (
    id          INTEGER PRIMARY KEY,
    role_name   TEXT    NOT NULL UNIQUE -- Contoh: 'super_admin', 'kasir', 'operator'
);

INSERT INTO roles VALUES(1,'super_admin');
INSERT INTO roles VALUES(2,'kasir');
INSERT INTO roles VALUES(3,'operator');

CREATE TABLE admins (
    id            INTEGER  PRIMARY KEY,
    role_id       INTEGER,
    username      TEXT     NOT NULL UNIQUE,
    password_hash TEXT     NOT NULL, -- Simpan password yang sudah di-bcrypt/hash
    nama_lengkap  TEXT     NOT NULL,
    is_active     INTEGER  DEFAULT 1, -- 1: Aktif, 0: Non-aktif
    last_login    DATETIME,
    created_at    DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (role_id) REFERENCES roles (id)
);

CREATE TABLE orders (
    id INTEGER PRIMARY KEY, 
    order_number TEXT UNIQUE NOT NULL, 
    customer_id INTEGER REFERENCES konsumen (id), 
    customer_name TEXT NOT NULL, 
    total_amount REAL DEFAULT 0,
    status TEXT DEFAULT 'pending', 
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    admin_id INTEGER REFERENCES admins (id) ON DELETE RESTRICT ON UPDATE CASCADE NOT NULL, 
    CONSTRAINT idx_order_admin UNIQUE (order_number COLLATE NOCASE, admin_id)
);

CREATE TABLE kategori_transaksi (
    id INTEGER PRIMARY KEY NOT NULL, 
    nama TEXT UNIQUE NOT NULL, 
    tipe TEXT CHECK (tipe IN ('pemasukan', 'pengeluaran'))
);

CREATE TABLE finishing_services (
    id INTEGER PRIMARY KEY, 
    name TEXT NOT NULL, 
    price_per_unit INTEGER DEFAULT 0
);

CREATE TABLE transaksi (
    id INTEGER PRIMARY KEY NOT NULL, 
    admin_id INTEGER REFERENCES admins (id) 
                    ON DELETE RESTRICT 
                    ON UPDATE CASCADE NOT NULL, 
    kategori_id INTEGER REFERENCES kategori_transaksi (id) 
                    ON DELETE RESTRICT 
                    ON UPDATE CASCADE, 
    deskripsi TEXT, 
    jumlah INTEGER NOT NULL, 
    tanggal TEXT NOT NULL DEFAULT (CURRENT_DATE)
);

CREATE TABLE payments (
    id INTEGER PRIMARY KEY NOT NULL, 
    paytime DATETIME NOT NULL DEFAULT (CURRENT_TIMESTAMP), 
    admin_id INTEGER REFERENCES admins (id) 
                    ON DELETE RESTRICT 
                    ON UPDATE CASCADE NOT NULL, 
    order_id INTEGER REFERENCES orders (id) 
                    ON DELETE RESTRICT 
                    ON UPDATE CASCADE NOT NULL, 
    amount INTEGER NOT NULL CHECK (amount > 0), 
    method TEXT NOT NULL CHECK (method IN ("cash", "transfer")) DEFAULT ('cash'), 
    transfer_acc TEXT, 
    transfer_ver BOOL DEFAULT (0)
);

CREATE UNIQUE INDEX "" ON products (sku COLLATE NOCASE);
CREATE INDEX idx_admin_tanggal ON transaksi (tanggal, admin_id);

COMMIT;
